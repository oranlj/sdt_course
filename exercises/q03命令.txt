mkdir q03
cd q03
git init

# main初始提交
echo "mode=normal" > config.txt
git add config.txt
git commit -m "initial commit: mode normal"

# feature‑a
git checkout -b feature-a
echo "mode=safe" > config.txt
git add config.txt
git commit -m "feature‑a: set mode safe"

# 回到main创建feature‑b
git checkout main
git checkout -b feature-b
echo "mode=fast" > config.txt
git add config.txt
git commit -m "feature‑b: set mode fast"

# 回到main，合并
git checkout main
git merge feature-a

# 合并feature‑b产生冲突
git merge feature-b

# 手动解决冲突
cat > config.txt <<'EOF'
mode=safe
note=reviewed
EOF

git add config.txt
git commit -m "merge feature‑b, resolve conflict: keep safe, add note"

git status
git log --all --graph --decorate --oneline
